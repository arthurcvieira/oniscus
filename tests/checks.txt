# Commands run on the serial console after the desktop session started
cat /etc/os-release
systemctl --failed --no-pager
sudo journalctl -b --no-pager -u 'ly@*' | tail -n 30
sudo ls -l /etc/ly/
grep -vE '^(#|$)' /etc/ly/config.ini | head -n 40
pgrep -a i3; pgrep -a xfce4-panel; pgrep -a polybar; pgrep -a plank; pgrep -a picom
cat ~/.cache/polybar.log
ls -l ~/.config/polybar/
export DISPLAY=:0 XAUTHORITY=/run/user/$(id -u)/lyxauth
timeout 5 polybar xi3 2>&1 | head -n 20
ps -u live -o pid,args --forest | grep -v ' \[' | tail -n 60
i3-msg -t get_tree | python3 -c 'import json,sys; t=json.load(sys.stdin); w=lambda n,d=0: [print("  "*d+str(n.get("type"))+" "+str(n.get("name"))+" "+str((n.get("window_properties") or {}).get("class"))+" "+str(n["rect"])) or [w(c,d+1) for c in n["nodes"]+n["floating_nodes"]] for _ in [0]]; w(t)'
xfconf-query -c xfce4-panel -lv | head -n 40
cat /etc/dconf/profile/user; ls -l /etc/dconf/db/ /etc/dconf/db/local.d/
gsettings get net.launchpad.plank.dock.settings:/net/launchpad/plank/docks/dock1/ position
gsettings get net.launchpad.plank.dock.settings:/net/launchpad/plank/docks/dock1/ hide-mode
ls /usr/lib64/gio/modules/; dconf dump /net/launchpad/plank/
systemctl --user --failed --no-pager
env | grep -E "^(DISPLAY|XAUTHORITY|XDG_)"; systemctl --user show-environment | grep -E "^(DISPLAY|XAUTHORITY)"
ls /usr/share/xsessions/ /usr/share/oniscus/xsessions/
ls /usr/lib64/xfce4/panel/plugins/ /usr/share/xfce4/panel/plugins/
rpm -ql xfce4-power-manager | grep -i plugin
ls -la ~/.local/state/ ~/.cache/ 2>/dev/null; tail -n 40 ~/.local/state/ly-session.log ~/.xsession-errors 2>/dev/null
journalctl --user -b --no-pager -p warning | tail -n 40
sudo journalctl -b --no-pager -p err | tail -n 40
# Empty workspace, so Plank's intelligent hide shows the dock in the last screenshot
i3-msg workspace 9; sleep 3
# The compositor should still be running with the Oniscus config
picom --version; pgrep -a picom
